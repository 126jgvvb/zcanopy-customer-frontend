```utils/format.ts
import type { ListingType } from '../types/property';

const compact = new Intl.NumberFormat('en-US', { maximumFractionDigits: 3 });

export function formatPrice(price: number, type: ListingType): string {
  if (type === 'rent') return `$${price.toLocaleString('en-US')}/mo`;
  if (price >= 1_000_000) return `$${compact.format(price / 1_000_000)}M`;
  return `$${Math.round(price / 1000)}K`;
}

export function formatSqft(sqft: number): string {
  return `${sqft.toLocaleString('en-US')} sq ft`;
}
```

```utils/filters.ts
import { budgetOptions, householdOptions, lifestyleOptions } from '../data/wizardOptions';
import type { Filters, Property } from '../types/property';
import { formatPrice } from './format';

export const defaultFilters: Filters = {
  listingType: 'any',
  city: 'any',
  maxPrice: null,
  minBeds: 0,
  kinds: [],
  lifestyle: [],
  mustHaves: [],
  sort: 'match',
};

export function toggleInArray<T>(list: T[], item: T): T[] {
  return list.includes(item) ? list.filter((i) => i !== item) : [...list, item];
}

export function filterProperties(list: Property[], f: Filters): Property[] {
  return list.filter(
    (p) =>
      (f.listingType === 'any' || p.listingType === f.listingType) &&
      (f.city === 'any' || p.city === f.city) &&
      (f.listingType === 'any' || f.maxPrice === null || p.price <= f.maxPrice) &&
      p.beds >= f.minBeds &&
      (f.kinds.length === 0 || f.kinds.includes(p.kind)) &&
      f.mustHaves.every((a) => p.amenities.includes(a)),
  );
}

export function matchScore(p: Property, f: Filters): number {
  let score = 64;
  if (f.lifestyle.length > 0) {
    const overlap = f.lifestyle.filter((l) => p.lifestyle.includes(l)).length;
    score += Math.round((28 * overlap) / f.lifestyle.length);
  } else {
    score += 20;
  }
  if (f.kinds.length > 0 && f.kinds.includes(p.kind)) score += 4;
  if (f.city !== 'any' && p.city === f.city) score += 3;
  if (f.mustHaves.length > 0) score += 2;
  return Math.min(score, 99);
}

export function matchReasons(p: Property, f: Filters): string[] {
  const reasons: string[] = [];
  const lifestyle = lifestyleOptions.filter((o) => f.lifestyle.includes(o.value) && p.lifestyle.includes(o.value));
  if (lifestyle.length) reasons.push(`Matches your vibe: ${lifestyle.map((l) => l.label.toLowerCase()).join(', ')}`);
  if (f.maxPrice !== null && f.listingType === p.listingType) {
    const diff = f.maxPrice - p.price;
    reasons.push(diff > 0 ? `${formatPrice(diff, p.listingType)} under your budget` : 'Right at your budget');
  }
  if (f.mustHaves.length) reasons.push(`Has all your must-haves: ${f.mustHaves.join(', ')}`);
  if (f.minBeds > 0) reasons.push(`${p.beds} bedrooms for your household`);
  if (f.city !== 'any') reasons.push(`In ${p.neighborhood}, ${p.city}`);
  return reasons;
}

export function sortProperties(list: Property[], f: Filters): Property[] {
  const sorted = [...list];
  switch (f.sort) {
    case 'price-asc':
      return sorted.sort((a, b) => a.price - b.price);
    case 'price-desc':
      return sorted.sort((a, b) => b.price - a.price);
    case 'newest':
      return sorted.sort((a, b) => a.listedDaysAgo - b.listedDaysAgo);
    default:
      return sorted.sort((a, b) => matchScore(b, f) - matchScore(a, f) || a.listedDaysAgo - b.listedDaysAgo);
  }
}

export function budgetLabel(f: Filters): string {
  if (f.listingType === 'any' || f.maxPrice === null) return 'No limit';
  return budgetOptions[f.listingType].find((o) => o.value === f.maxPrice)?.label ?? formatPrice(f.maxPrice, f.listingType);
}

export function householdLabel(minBeds: number): string {
  return householdOptions.find((o) => o.minBeds === minBeds)?.label ?? `${minBeds}+ bedrooms`;
}

export interface ActiveChip {
  key: string;
  label: string;
  clear: Partial<Filters>;
}

export function getActiveChips(f: Filters): ActiveChip[] {
  const chips: ActiveChip[] = [];
  if (f.listingType !== 'any')
    chips.push({ key: 'type', label: f.listingType === 'buy' ? 'For sale' : 'For rent', clear: { listingType: 'any', maxPrice: null } });
  if (f.city !== 'any') chips.push({ key: 'city', label: f.city, clear: { city: 'any' } });
  if (f.listingType !== 'any' && f.maxPrice !== null) chips.push({ key: 'budget', label: budgetLabel(f), clear: { maxPrice: null } });
  if (f.minBeds > 0) chips.push({ key: 'beds', label: `${f.minBeds}+ beds`, clear: { minBeds: 0 } });
  f.kinds.forEach((k) => chips.push({ key: `kind-${k}`, label: k, clear: { kinds: f.kinds.filter((x) => x !== k) } }));
  f.mustHaves.forEach((a) => chips.push({ key: `amenity-${a}`, label: a, clear: { mustHaves: f.mustHaves.filter((x) => x !== a) } }));
  return chips;
}
```

```contexts/AppContext.tsx
import React, { createContext, useCallback, useContext, useEffect, useMemo, useState } from 'react';
import { defaultFilters } from '../utils/filters';
import type { Filters } from '../types/property';

interface AppContextValue {
  saved: string[];
  isSaved: (id: string) => boolean;
  toggleSave: (id: string) => void;
  filters: Filters;
  setFilters: (f: Filters) => void;
  updateFilters: (patch: Partial<Filters>) => void;
  resetFilters: () => void;
  hasPreferences: boolean;
  setHasPreferences: (v: boolean) => void;
  wizardOpen: boolean;
  openWizard: () => void;
  closeWizard: () => void;
}

const STORAGE_KEY = 'zcanopy:saved';
const AppContext = createContext<AppContextValue | null>(null);

export function AppProvider({ children }: { children: React.ReactNode }) {
  const [saved, setSaved] = useState<string[]>(() => {
    try {
      const raw = localStorage.getItem(STORAGE_KEY);
      return raw ? (JSON.parse(raw) as string[]) : [];
    } catch {
      return [];
    }
  });
  const [filters, setFilters] = useState<Filters>(defaultFilters);
  const [hasPreferences, setHasPreferences] = useState(false);
  const [wizardOpen, setWizardOpen] = useState(false);

  useEffect(() => {
    try {
      localStorage.setItem(STORAGE_KEY, JSON.stringify(saved));
    } catch {
      /* storage unavailable */
    }
  }, [saved]);

  const toggleSave = useCallback((id: string) => {
    setSaved((s) => (s.includes(id) ? s.filter((x) => x !== id) : [...s, id]));
  }, []);

  const value = useMemo<AppContextValue>(
    () => ({
      saved,
      isSaved: (id) => saved.includes(id),
      toggleSave,
      filters,
      setFilters,
      updateFilters: (patch) => setFilters((f) => ({ ...f, ...patch })),
      resetFilters: () => {
        setFilters(defaultFilters);
        setHasPreferences(false);
      },
      hasPreferences,
      setHasPreferences,
      wizardOpen,
      openWizard: () => setWizardOpen(true),
      closeWizard: () => setWizardOpen(false),
    }),
    [saved, toggleSave, filters, hasPreferences, wizardOpen],
  );

  return <AppContext.Provider value={value}>{children}</AppContext.Provider>;
}

export function useApp(): AppContextValue {
  const ctx = useContext(AppContext);
  if (!ctx) throw new Error('useApp must be used inside AppProvider');
  return ctx;
}
```

```hooks/useTourPlayback.ts
import { useCallback, useEffect, useState } from 'react';

interface Options {
  duration?: number;
  active?: boolean;
}

export function useTourPlayback(count: number, { duration = 5000, active = true }: Options = {}) {
  const [index, setIndex] = useState(0);
  const [elapsed, setElapsed] = useState(0);
  const [playing, setPlaying] = useState(true);

  useEffect(() => {
    if (!active || !playing) return;
    const id = window.setInterval(() => setElapsed((e) => e + 100), 100);
    return () => window.clearInterval(id);
  }, [active, playing]);

  useEffect(() => {
    if (elapsed >= duration) {
      setIndex((i) => (i + 1) % count);
      setElapsed(0);
    }
  }, [elapsed, duration, count]);

  useEffect(() => {
    if (!active) {
      setIndex(0);
      setElapsed(0);
      setPlaying(true);
    }
  }, [active]);

  const goTo = useCallback(
    (i: number) => {
      setIndex(((i % count) + count) % count);
      setElapsed(0);
    },
    [count],
  );

  const toggle = useCallback(() => setPlaying((p) => !p), []);

  return {
    index,
    progress: Math.min(elapsed / duration, 1),
    playing,
    toggle,
    goTo,
    next: () => goTo(index + 1),
    prev: () => goTo(index - 1),
  };
}
```

```pages/Reels.tsx
import React, { useEffect, useMemo, useRef, useState } from 'react';
import { useSearchParams } from 'react-router-dom';
import { ChevronDownIcon, ChevronUpIcon } from 'lucide-react';
import { ReelItem } from '../components/ReelItem';
import { useApp } from '../contexts/AppContext';
import { properties } from '../data/properties';
import { filterProperties, matchReasons, matchScore, sortProperties } from '../utils/filters';

export function Reels() {
  const { filters, hasPreferences } = useApp();
  const [params] = useSearchParams();
  const containerRef = useRef<HTMLDivElement>(null);

  const list = useMemo(() => {
    if (!hasPreferences) return properties;
    const matched = sortProperties(filterProperties(properties, filters), { ...filters, sort: 'match' });
    const rest = properties.filter((p) => !matched.includes(p));
    return [...matched, ...rest];
  }, [filters, hasPreferences]);

  const [activeId, setActiveId] = useState(params.get('start') ?? list[0]?.id);
  const activeIndex = list.findIndex((p) => p.id === activeId);

  useEffect(() => {
    const root = containerRef.current;
    if (!root) return;
    const observer = new IntersectionObserver(
      (entries) => {
        entries.forEach((e) => {
          if (e.isIntersecting) setActiveId((e.target as HTMLElement).dataset.id);
        });
      },
      { root, threshold: 0.6 },
    );
    root.querySelectorAll('.reel').forEach((el) => observer.observe(el));
    return () => observer.disconnect();
  }, [list]);

  useEffect(() => {
    const start = params.get('start');
    if (!start) return;
    containerRef.current?.querySelector(`[data-id="${start}"]`)?.scrollIntoView({ block: 'start' });
  }, [params]);

  const step = (dir: 1 | -1) => {
    const root = containerRef.current;
    if (root) root.scrollBy({ top: dir * root.clientHeight, behavior: 'smooth' });
  };

  return (
    <main className="relative bg-ink" style={{ height: 'calc(100dvh - 4rem)' }}>
      <h1 className="sr-only">Property reels</h1>
      <div ref={containerRef} className="h-full snap-y snap-mandatory overflow-y-scroll scrollbar-none">
        {list.map((p) => {
          const matched = hasPreferences && filterProperties([p], filters).length > 0;
          return (
            <ReelItem
              key={p.id}
              property={p}
              active={p.id === activeId}
              score={matched ? matchScore(p, filters) : null}
              reasons={matched ? matchReasons(p, filters) : []}
            />
          );
        })}
      </div>

      <div className="absolute right-6 top-1/2 hidden -translate-y-1/2 flex-col items-center gap-2 lg:flex">
        <button
          type="button"
          onClick={() => step(-1)}
          disabled={activeIndex <= 0}
          aria-label="Previous home"
          className="flex h-11 w-11 items-center justify-center rounded-full bg-white/10 text-white transition-colors duration-150 hover:bg-white/20 disabled:opacity-30"
        >
          <ChevronUpIcon className="h-5 w-5" aria-hidden="true" />
        </button>
        <span className="text-xs font-medium tabular-nums text-white/60">
          {activeIndex + 1}/{list.length}
        </span>
        <button
          type="button"
          onClick={() => step(1)}
          disabled={activeIndex >= list.length - 1}
          aria-label="Next home"
          className="flex h-11 w-11 items-center justify-center rounded-full bg-white/10 text-white transition-colors duration-150 hover:bg-white/20 disabled:opacity-30"
        >
          <ChevronDownIcon className="h-5 w-5" aria-hidden="true" />
        </button>
      </div>
    </main>
  );
}
```
